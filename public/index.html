              import { Injectable } from '@nestjs/common';
import { OrderRepository } from './order.repository';
import { CreateOrderDto, OrderStatus } from './order.types';

/**
 * 주문 생성 및 상태 전이를 담당한다.
 * 결제 승인 전에는 PENDING, 승인 후 CONFIRMED로 전환한다.
 */
@Injectable()
export class OrderService {
  constructor(private readonly repo: OrderRepository) {}

  async create(dto: CreateOrderDto) {
    const total = dto.items.reduce((sum, i) => sum + i.price * i.quantity, 0);
    if (total <= 0) throw new Error('주문 금액이 올바르지 않습니다');
    return this.repo.save({ ...dto, total, status: OrderStatus.PENDING });
  }

  async confirm(orderId: string) {
    const order = await this.repo.findById(orderId);
    if (!order) throw new Error(`주문을 찾을 수 없습니다: ${orderId}`);
    // TODO: 재고 차감은 트랜잭션으로 묶을 것
    return this.repo.update(orderId, { status: OrderStatus.CONFIRMED });
  }
}
